Key syntax to understand

/* Declare in :root = global scope */ :root { --color-accent: #c8f557; --ease-out-expo: cubic-bezier(0.16, 1, 0.3, 1); } /* Use with var() */ .button { background: var(--color-accent); transition: transform var(--duration-base) var(--ease-out-expo); } /* Scope override — component gets its own value */ .card--dark { --color-surface: #1a1a1a; /* only affects this card */ } /* JS can read AND write them */ document.documentElement.style.setProperty('--color-accent', '#ff6b35');
